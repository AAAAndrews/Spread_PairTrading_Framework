# 数据库锁定快速修复 - Notebook版本

在notebook中运行此代码块来修复数据库锁定问题

## 使用方法

将以下代码复制到一个新的notebook单元格中运行：

```python
# ============================================================
# 快速修复数据库锁定问题
# ============================================================
import sqlite3
import os
import time
from pathlib import Path

DB_PATH = "data/trading_data.db"

def quick_fix_database_lock():
    """快速修复数据库锁定"""
    print("🔧 开始修复数据库锁定问题...")
    print("=" * 60)
    
    # 1. 检查临时文件
    print("\n[1/5] 检查临时文件...")
    temp_files = [
        DB_PATH + "-wal",
        DB_PATH + "-shm", 
        DB_PATH + "-journal"
    ]
    
    found_locks = []
    for temp_file in temp_files:
        if os.path.exists(temp_file):
            found_locks.append(temp_file)
            print(f"  ⚠️  发现: {temp_file}")
    
    if not found_locks:
        print("  ✅ 没有临时文件")
    
    # 2. 尝试关闭连接
    print("\n[2/5] 尝试关闭现有连接...")
    try:
        conn = sqlite3.connect(DB_PATH, timeout=5.0)
        conn.execute("PRAGMA wal_checkpoint(FULL)")
        conn.close()
        print("  ✅ 连接已关闭")
    except Exception as e:
        print(f"  ⚠️  {e}")
    
    # 3. 删除临时文件
    if found_locks:
        print("\n[3/5] 删除临时文件...")
        for temp_file in found_locks:
            try:
                os.remove(temp_file)
                print(f"  ✅ 已删除: {os.path.basename(temp_file)}")
            except Exception as e:
                print(f"  ❌ 无法删除: {e}")
        time.sleep(1)
    else:
        print("\n[3/5] 跳过（无需删除）")
    
    # 4. 测试连接
    print("\n[4/5] 测试数据库连接...")
    try:
        conn = sqlite3.connect(DB_PATH, timeout=10.0)
        cursor = conn.cursor()
        cursor.execute("SELECT COUNT(*) FROM sqlite_master WHERE type='table'")
        table_count = cursor.fetchone()[0]
        print(f"  ✅ 连接成功！找到 {table_count} 个表")
        conn.close()
    except Exception as e:
        print(f"  ❌ 连接失败: {e}")
        return False
    
    # 5. 应用推荐设置
    print("\n[5/5] 应用推荐设置...")
    try:
        conn = sqlite3.connect(DB_PATH, timeout=10.0)
        conn.execute("PRAGMA journal_mode = WAL")
        conn.execute("PRAGMA synchronous = NORMAL")
        conn.execute("PRAGMA busy_timeout = 5000")
        conn.close()
        print("  ✅ 设置已应用")
    except Exception as e:
        print(f"  ⚠️  {e}")
    
    print("\n" + "=" * 60)
    print("✅ 修复完成！现在可以重新运行你的代码了")
    return True

# 执行修复
quick_fix_database_lock()
```

## 预防措施

在代码中添加以下内容来预防数据库锁定：

```python
# 1. 使用context manager
with sqlite3.connect(DB_PATH) as conn:
    cursor = conn.cursor()
    # 你的操作
    conn.commit()
# 连接自动关闭

# 2. 批量插入而非逐行插入
# 不好的做法：
for row in data:
    db.insert_price_data(row, ...)

# 好的做法：
df_to_insert = pd.DataFrame(data)
db.insert_price_data(df_to_insert, ...)

# 3. 定期提交事务
conn = sqlite3.connect(DB_PATH)
for i, row in enumerate(data):
    # 插入数据
    if i % 1000 == 0:  # 每1000行提交一次
        conn.commit()
conn.commit()
conn.close()
```

## 常见原因

1. **未关闭的连接**：程序异常退出时连接未关闭
2. **并发写入**：多个进程同时写入数据库
3. **长事务**：事务运行时间过长，阻塞其他操作
4. **WAL模式问题**：WAL临时文件未正确清理

## 终极方案

如果所有方法都无效，备份并重建数据库：

```python
import shutil
import sqlite3

# 1. 备份数据库
shutil.copy('data/trading_data.db', 'data/trading_data_backup.db')

# 2. 导出数据
old_conn = sqlite3.connect('data/trading_data_backup.db')
new_conn = sqlite3.connect('data/trading_data_new.db')

# 复制表结构和数据
for line in old_conn.iterdump():
    new_conn.execute(line)

new_conn.commit()
old_conn.close()
new_conn.close()

# 3. 替换数据库
os.remove('data/trading_data.db')
shutil.move('data/trading_data_new.db', 'data/trading_data.db')
```
